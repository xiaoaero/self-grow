```javascript
/** 根据累计总经验计算等级（上限200） */
function levelFromTotalExp(totalExp) {
    if (totalExp <= 0) return 1;
    let level = 1;
    let cumulative = 0;
    while (level < 200) {
        const need = expForLevel(level);
        if (cumulative + need > totalExp) break;
        cumulative += need;
        level++;
    }
    return Math.min(level, 200);
}
```

```python
    """根据累计总经验值计算当前等级（上限 200 级）"""
    if total_exp <= 0:

    level = 1
    cumulative = 0

        level += 1
    return min(level, 200)
```
